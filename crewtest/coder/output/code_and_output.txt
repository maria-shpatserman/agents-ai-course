# This program calculates the value of the series 1 - 1/3 + 1/5 - 1/7 + ... 
# for the first 10,000 terms and then multiplies the result by 4.

def calculate_series(terms):
    total = 0
    for n in range(terms):
        # The denominator is generated as 2*n + 1 to get odd numbers
        denominator = 2 * n + 1
        # Alternating between adding and subtracting:
        if n % 2 == 0:
            total += 1 / denominator
        else:
            total -= 1 / denominator
    return total * 4  # Multiply the result by 4, as required

# Define the number of terms
number_of_terms = 10000

# Calculate the series
result = calculate_series(number_of_terms)

# Output the result
print("The result of the series after multiplying by 4 is:", result)
